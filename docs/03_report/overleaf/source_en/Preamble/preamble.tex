% Language switch: \thesisvitrue = Vietnamese edition, \thesisvifalse = English edition.
\newif\ifthesisvi
\thesisvifalse
\ifthesisvi
  \usepackage[T1,T5]{fontenc}          % T5 (Vietnamese) is the default encoding
  \usepackage[english,vietnamese]{babel}
  % Heading names written directly in Unicode so PDF bookmarks keep their diacritics
  \addto\captionsvietnamese{\renewcommand{\listtablename}{Danh sách bảng}\renewcommand{\bibname}{Tài liệu tham khảo}}
  \newcommand{\vi}[1]{#1}
  % Vietnamese names for headings that the class hard-codes in English.
  % Also add each front-matter chapter* to TOC + PDF bookmarks (same chapter
  % level as LOF/LOT via tocbibind). \phantomsection before \addcontentsline
  % so hyperlinks land on the correct page.
  \usepackage{etoolbox}
  \newcommand{\vivqaFrontmatterTocEntry}[1]{%
    \phantomsection
    \addcontentsline{toc}{chapter}{#1}%
  }
  % \texorpdfstring: avoids hyperref "Token not allowed" from \centering/\Large in bookmarks
  \patchcmd{\declaration}
    {\chapter*{\centering \Large Declaration}}
    {\vivqaFrontmatterTocEntry{Lời cam đoan}\chapter*{\texorpdfstring{\centering \Large Lời cam đoan}{Lời cam đoan}}}
    {}{\PackageWarning{vivqa}{declaration toc patch failed}}
  \patchcmd{\acknowledgements}
    {\chapter*{\centering \Large Acknowledgements}}
    {\vivqaFrontmatterTocEntry{Lời cảm ơn}\chapter*{\texorpdfstring{\centering \Large Lời cảm ơn}{Lời cảm ơn}}}
    {}{\PackageWarning{vivqa}{acknowledgements toc patch failed}}
  \patchcmd{\abstract}
    {\chapter*{\centering \Large Abstract}}
    {\vivqaFrontmatterTocEntry{Tóm tắt}\chapter*{\texorpdfstring{\centering \Large Tóm tắt}{Tóm tắt}}}
    {}{\PackageWarning{vivqa}{abstract toc patch failed}}
  \renewcommand{\nomname}{Danh mục từ viết tắt}
  \renewcommand{\nomgroup}[1]{\ifthenelse{\equal{#1}{Z}}{\item[\textbf{Từ viết tắt}]}{}}
\else
  % English edition. T5 is loaded last, so it is the default font encoding: the
  % Vietnamese examples (questions, answers, OCR text, names) render directly,
  % while all body text, captions and headings are English (babel: english).
  \usepackage[T1,T5]{fontenc}
  \usepackage[vietnamese,english]{babel}
  \addto\captionsenglish{\renewcommand{\bibname}{References}}
  \newcommand{\vi}[1]{\foreignlanguage{vietnamese}{#1}}
  % Let TeX stretch interword spaces a little more instead of overfilling lines
  % that contain long unbreakable names such as Qwen2.5-VL-3B-Instruct.
  \setlength{\emergencystretch}{3em}
  % Add each front-matter chapter* to the TOC and PDF bookmarks.
  \usepackage{etoolbox}
  \newcommand{\vivqaFrontmatterTocEntry}[1]{%
    \phantomsection
    \addcontentsline{toc}{chapter}{#1}%
  }
  \patchcmd{\declaration}
    {\chapter*{\centering \Large Declaration}}
    {\vivqaFrontmatterTocEntry{Declaration}\chapter*{\texorpdfstring{\centering \Large Declaration}{Declaration}}}
    {}{\PackageWarning{vivqa}{declaration toc patch failed}}
  \patchcmd{\acknowledgements}
    {\chapter*{\centering \Large Acknowledgements}}
    {\vivqaFrontmatterTocEntry{Acknowledgements}\chapter*{\texorpdfstring{\centering \Large Acknowledgements}{Acknowledgements}}}
    {}{\PackageWarning{vivqa}{acknowledgements toc patch failed}}
  \patchcmd{\abstract}
    {\chapter*{\centering \Large Abstract}}
    {\vivqaFrontmatterTocEntry{Abstract}\chapter*{\texorpdfstring{\centering \Large Abstract}{Abstract}}}
    {}{\PackageWarning{vivqa}{abstract toc patch failed}}
  \renewcommand{\nomname}{List of Abbreviations}
  \renewcommand{\nomgroup}[1]{\ifthenelse{\equal{#1}{Z}}{\item[\textbf{Abbreviations}]}{}}
\fi
